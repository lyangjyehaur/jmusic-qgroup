const familiarityWeight = { '本命': 4, '常聽': 3, '聽過一些': 2, '只聽過幾首': 1 };

export interface MatchResult {
  user: User;
  common: { term: Term; mine: Selection; theirs: Selection }[];
  strict: number;
  favorite: number;
  rare: number;
  score: number;
  match: number;
}

export function calculateMatch(currentUser: User, user: User, terms: Term[]): MatchResult {
  const common = currentUser.selections.flatMap((mine) => {
    const theirs = user.selections.find((s) => s.termId === mine.termId);
    const term = terms.find((t) => t.id === mine.termId);
    return theirs && term ? [{ term, mine, theirs }] : [];
  });
  const strict = common.filter((item) => item.mine.familiarity === item.theirs.familiarity).length;
  const favorite = common.filter((item) => item.mine.familiarity === '本命' && item.theirs.familiarity === '本命').length;
  const rare = common.reduce((sum, item) => sum + Math.max(0, 36 - item.term.users), 0);
  const score = common.length * 12 + strict * 4 + favorite * 8 + rare * 0.3;
  const match = Math.min(99, Math.round(common.length ? 28 + (common.length / Math.max(currentUser.selections.length, user.selections.length)) * 48 + strict * 4 + favorite * 6 : 0));
  return { user, common, strict, favorite, rare, score, match };
}
