function Card({ name, purpose, usage, where, children, note }: {
  name: string; purpose: string; usage: Usage; where?: string; children: ReactNode; note?: string;
}) {
  return <article className="ui-kit-card">
    <span className={usage === '已用於原型' ? 'ui-kit-tag used' : 'ui-kit-tag'}>{usage}{where ? `｜${where}` : ''}</span>
    <h3>{name}</h3>
    <p>{purpose}</p>
    <div className="ui-kit-demo">{children}</div>
    {note && <p className="ui-kit-note">{note}</p>}
  </article>;
}

function Group({ index, title, description, children }: {
  index: string; title: string; description: string; children: ReactNode;
}) {

    <div className="section-heading"><span>{index}</span><div><h2>{title}</h2><p>{description}</p></div></div>
    <div className="ui-kit-grid">{children}</div>
  </section>;
}
