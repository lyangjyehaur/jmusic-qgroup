import { useState } from 'react';
import { Link, useSearchParams } from 'react-router-dom';
import { AlertTriangle, ArrowRight, Check, ChevronRight, CircleUserRound, KeyRound, Laptop, Languages, LogOut, Music2, Palette, Plus, QrCode, RotateCcw, ShieldCheck, Smartphone, Trash2, UserRoundX, UsersRound } from 'lucide-react';
import { useApp } from '../services/AppStore';
import { Avatar, Badge, Button, Field, Modal, PageHeader, Select, Tabs, cn } from '../components/UI';
import { presetLabels, useYahhaPreset, type YahhaPreset } from '../theme/YahhaTheme';

export default function MePage() {
  const { currentUser, users, activities, savedSelections, switchUser, updateUser, saveSelections, resetPrototype, showToast } = useApp();
  const { preset, setPreset } = useYahhaPreset();
  const [params, setParams] = useSearchParams();
  const tab = params.get('tab') ?? 'overview';
  const [name, setName] = useState(currentUser.name);
  const [deleteMode, setDeleteMode] = useState<'music' | 'account' | null>(null);
  const [confirmText, setConfirmText] = useState('');
  const myActivities = activities.filter((a) => a.organizerId === currentUser.id || a.participants.some((p) => p.userId === currentUser.id));
  const drafts = activities.filter((a) => a.organizerId === currentUser.id && a.status === '草稿');

  return <div className="page me-page">
    <PageHeader title="我的" description={`${currentUser.name}・${currentUser.profileId}`} actions={<Link to="/me/music"><Button><Music2 size={16} />編輯我的成分</Button></Link>} />
    <section className="profile-banner"><Avatar user={currentUser} size="xl" /><div><h2>{currentUser.name}</h2><p>{currentUser.message || '尚未填寫公開留言'}</p><span>{currentUser.profileId}・QQ 已綁定（虛構）</span></div><div className="profile-numbers"><Link to="/me/music"><strong>{savedSelections.length}</strong><span>音樂成分</span></Link><Link to="/activities"><strong>{myActivities.length}</strong><span>我的活動</span></Link><Link to="/me?tab=drafts"><strong>{drafts.length}</strong><span>活動草稿</span></Link></div></section>
    <Tabs value={tab} onChange={(value) => setParams(value === 'overview' ? {} : { tab: value })} items={[{ value: 'overview', label: '總覽' }, { value: 'drafts', label: '活動草稿', count: drafts.length }, { value: 'account', label: '帳號與裝置' }, { value: 'privacy', label: '隱私與外觀' }, { value: 'danger', label: '資料管理' }]} />

    {tab === 'overview' && <div className="me-grid"><section className="me-main"><div className="section-title-row"><div><h2>我的活動</h2><p>近期報名與發起的安排</p></div><Link to="/activities" className="text-link">全部活動<ArrowRight size={14} /></Link></div><div className="me-activity-list">{myActivities.filter((a) => !['已結束', '已歸檔'].includes(a.status)).slice(0, 4).map((activity) => { const mine = activity.participants.find((p) => p.userId === currentUser.id); return <Link to={`/activities/${activity.id}`} key={activity.id}><span className="me-activity-date">{new Date(activity.start).getMonth() + 1}/{new Date(activity.start).getDate()}</span><span><strong>{activity.name}</strong><small>{activity.location}</small></span><Badge tone={mine?.status === '正式' ? 'green' : 'neutral'}>{mine?.status ?? activity.status}</Badge><ChevronRight size={16} /></Link>; })}</div></section><aside className="settings-links"><h2>常用設定</h2><Link to="/me/music"><Music2 size={18} /><span>音樂成分與詞條<small>{savedSelections.length} 項成分</small></span><ChevronRight size={15} /></Link><Link to="/ledger/a001"><QrCode size={18} /><span>收款碼與帳單<small>模擬支付資料</small></span><ChevronRight size={15} /></Link><button onClick={() => setParams({ tab: 'account' })}><KeyRound size={18} /><span>QQ 綁定與裝置<small>帳號安全與授權</small></span><ChevronRight size={15} /></button><button onClick={() => setParams({ tab: 'privacy' })}><Palette size={18} /><span>外觀與語言<small>舒適密度・繁體中文</small></span><ChevronRight size={15} /></button></aside></div>}

    {tab === 'drafts' && <section className="tab-panel"><div className="section-title-row"><div><h2>活動草稿</h2><p>草稿不開放報名，也不發送群通知</p></div><Link to="/activities/new"><Button><Plus size={16} />新增活動</Button></Link></div>{drafts.map((draft) => <article className="draft-row" key={draft.id}><div><Badge>草稿</Badge><h3>{draft.name}</h3><p>{draft.start.replace('T', ' ')}・{draft.location}</p></div><Link to={`/activities/new?edit=${draft.id}`}><Button variant="secondary" size="sm">繼續編輯</Button></Link><Link to={`/activities/${draft.id}`}><Button variant="text" size="sm">查看</Button></Link></article>)}{drafts.length === 0 && <p className="muted">目前沒有此帳號的草稿。切換帳號後會看到各自的資料。</p>}</section>}

    {tab === 'account' && <div className="settings-grid"><section className="form-section"><div className="section-heading"><span>帳號</span><div><h2>站內資料與 QQ 綁定</h2><p>全站沿用相同稱呼、成分與未讀狀態</p></div></div><Field label="站內稱呼"><input value={name} onChange={(e) => setName(e.target.value)} maxLength={30} /></Field><Button onClick={() => { updateUser(currentUser.id, { name: name.trim() || currentUser.name }); showToast('站內稱呼已更新'); }}>保存稱呼</Button><div className="binding-row"><div><strong>QQ 綁定</strong><span>{currentUser.qq}（虛構示例）</span></div><Badge tone="green">已綁定</Badge><Button variant="secondary" size="sm" onClick={() => showToast('重新授權流程已完成（模擬），未連接 QQ')}>重新授權</Button></div><div className="binding-row"><div><strong>收款碼</strong><span>非真實支付帳戶</span></div><Link to="/ledger/a001"><Button variant="secondary" size="sm">管理</Button></Link></div></section><section className="form-section"><div className="section-heading"><span>切換</span><div><h2>多帳號</h2><p>用於演示不同權限與資料狀態</p></div></div><div className="account-switch-list">{users.slice(0, 4).map((user) => <button key={user.id} className={cn(user.id === currentUser.id && 'active')} onClick={() => { switchUser(user.id); setName(user.name); showToast(`已切換為 ${user.name}`, 'info'); }}><Avatar user={user} /><span><strong>{user.name}</strong><small>{user.profileId}・{user.role}</small></span>{user.id === currentUser.id ? <Check size={17} /> : <span>切換</span>}</button>)}</div></section><section className="form-section span-full"><div className="section-heading"><span>裝置</span><div><h2>登入裝置</h2><p>移除裝置只影響此原型中的裝置清單</p></div></div><div className="device-list"><div><Laptop size={20} /><span><strong>MacBook Pro・此裝置</strong><small>WorkBuddy 瀏覽器・目前使用</small></span><Badge tone="green">目前</Badge></div><div><Smartphone size={20} /><span><strong>iPhone</strong><small>行動瀏覽器・上次使用 11-22</small></span><Button size="sm" variant="text" onClick={(e) => { e.currentTarget.closest('div')?.remove(); showToast('已移除裝置（本次畫面）', 'info'); }}>移除</Button></div></div></section></div>}

    {tab === 'privacy' && <div className="settings-grid"><section className="form-section"><div className="section-heading"><span>外觀</span><div><h2>外觀與語言</h2><p>設定保存在此瀏覽器</p></div></div><Field label="介面密度"><Select defaultValue={localStorage.getItem('gunju.density') ?? '舒適'} onChange={(e) => { localStorage.setItem('gunju.density', e.target.value); document.documentElement.dataset.density = e.target.value === '緊湊' ? 'compact' : 'comfortable'; showToast('介面密度已更新'); }}><option>舒適</option><option>緊湊</option></Select></Field><Field label="語言"><Select value="繁體中文" disabled><option>繁體中文</option></Select></Field><Field label="元件配色" hint="切換後全站與對話框同步更新"><Select value={preset} onChange={(e) => setPreset(e.target.value as YahhaPreset)}>{Object.entries(presetLabels).map(([key, label]) => <option key={key} value={key}>{label}</option>)}</Select></Field><Link to="/ui-kit" className="text-link">查看 32 個元件的用途對照</Link><div className="inline-alert"><Languages size={17} />原型介面固定使用繁體中文；元件庫內建簡體文案已於轉接層處理。</div></section><section className="form-section"><div className="section-heading"><span>隱私</span><div><h2>資料可見與同意</h2><p>成分刪除與帳號註銷分開處理</p></div></div>{([['messageVisible', '公開留言'], ['genderVisible', '公開性別'], ['qqVisible', '公開聯絡 QQ']] as const).map(([key, label]) => <label className="switch-row" key={key}><span><strong>{label}</strong><small>授權後才會出現在同好成分卡</small></span><input type="checkbox" checked={currentUser[key]} onChange={(e) => updateUser(currentUser.id, { [key]: e.target.checked })} /><i /></label>)}<label className="switch-row"><span><strong>隱私政策同意</strong><small>版本 2026-11・原型本機儲存</small></span><input type="checkbox" defaultChecked onChange={() => showToast('已更新隱私同意紀錄（模擬）')} /><i /></label></section></div>}

    {tab === 'danger' && <div className="settings-grid"><section className="form-section"><div className="section-heading"><span>備份</span><div><h2>原型資料</h2><p>清除本機修改，恢復初始示例</p></div></div><p className="muted">活動、成分、通知與款項修改目前僅保存在瀏覽器 localStorage。</p><Button variant="secondary" onClick={() => { if (window.confirm('確定恢復初始示例資料？所有本機修改都會被取代。')) resetPrototype(); }}><RotateCcw size={16} />恢復初始示例</Button></section><section className="form-section danger-zone"><div className="section-heading"><span>注意</span><div><h2>資料刪除</h2><p>兩種操作範圍不同，需分別確認</p></div></div><div><span><strong>刪除音樂成分</strong><small>只清空本人成分，保留帳號、活動與詞條</small></span><Button variant="secondary" onClick={() => setDeleteMode('music')}><Trash2 size={15} />刪除成分</Button></div><div><span><strong>註銷帳號</strong><small>包含帳號狀態與關聯資料；原型僅記錄模擬申請</small></span><Button variant="danger" onClick={() => setDeleteMode('account')}><UserRoundX size={15} />申請註銷</Button></div></section></div>}

    {deleteMode && <Modal title={deleteMode === 'music' ? '刪除我的音樂成分' : '申請帳號註銷'} description={deleteMode === 'music' ? '不會刪除帳號、活動或你建立的詞條。' : '這不是刪除音樂成分；原型只記錄模擬申請。'} onClose={() => { setDeleteMode(null); setConfirmText(''); }} footer={<><Button variant="secondary" onClick={() => setDeleteMode(null)}>取消</Button><Button variant="danger" disabled={confirmText !== (deleteMode === 'music' ? '刪除成分' : '註銷帳號')} onClick={() => { if (deleteMode === 'music') { saveSelections([]); showToast('音樂成分已清空，帳號仍保留', 'info'); } else showToast('帳號註銷申請已記錄（模擬），未操作正式帳號', 'info'); setDeleteMode(null); setConfirmText(''); }}>確認{deleteMode === 'music' ? '刪除成分' : '申請註銷'}</Button></>}><div className="inline-alert error"><AlertTriangle size={17} />請輸入「{deleteMode === 'music' ? '刪除成分' : '註銷帳號'}」以確認。</div><Field label="確認文字"><input value={confirmText} onChange={(e) => setConfirmText(e.target.value)} /></Field></Modal>}
  </div>;
}
