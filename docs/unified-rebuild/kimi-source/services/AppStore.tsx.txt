import { createContext, useCallback, useContext, useEffect, useMemo, useState, type ReactNode } from 'react';
import { initialActivities, initialBills, initialBotDeliveries, initialLedger, initialNotifications, initialTerms, initialUsers, initialVoteOptions } from '../data/mockData';
import type { Activity, AppNotification, Bill, BotDelivery, LedgerEntry, Participant, Selection, Term, User, VoteOption } from '../types';

const STORAGE_KEY = 'gunju.prototype.v1';
const CURRENT_USER_ID = 'u01';

interface PersistedState {
  currentUserId: string;
  activities: Activity[];
  users: User[];
  terms: Term[];
  draftSelections: Selection[];
  notifications: AppNotification[];
  bills: Bill[];
  ledger: LedgerEntry[];
  voteOptions: VoteOption[];
  botDeliveries: BotDelivery[];
}

interface ToastMessage { id: number; text: string; tone: 'success' | 'error' | 'info' }

interface AppContextValue extends PersistedState {
  currentUser: User;
  savedSelections: Selection[];
  toasts: ToastMessage[];
  showToast: (text: string, tone?: ToastMessage['tone']) => void;
  dismissToast: (id: number) => void;
  setDraftSelections: (selections: Selection[]) => void;
  saveSelections: (selections: Selection[]) => void;
  updateActivity: (id: string, updater: (activity: Activity) => Activity) => void;
  createActivity: (activity: Activity) => void;
  duplicateActivity: (id: string) => string;
  register: (activityId: string, participants: Pick<Participant, 'name' | 'userId' | 'companionOf' | 'feeFactor'>[], waitlistMode: '個別候補' | '整組候補') => void;
  updateParticipant: (activityId: string, participantId: string, patch: Partial<Participant>) => void;
  updateUser: (id: string, patch: Partial<User>) => void;
  switchUser: (id: string) => void;
  saveTerm: (term: Term) => void;
  deleteTerm: (id: string) => void;
  updateNotification: (id: string, patch: Partial<AppNotification>) => void;
  declareBill: (id: string) => void;
  verifyBill: (id: string) => void;
  setVote: (optionId: string, value: '可以' | '可能' | '不行') => void;
  finalizeVote: (optionId: string) => void;
  retryDelivery: (id: string) => void;
  resetPrototype: () => void;
}

const AppContext = createContext<AppContextValue | null>(null);

function freshState(): PersistedState {
  return {
    currentUserId: CURRENT_USER_ID,
    activities: initialActivities, users: initialUsers, terms: initialTerms,
    draftSelections: initialUsers[0].selections, notifications: initialNotifications,
    bills: initialBills, ledger: initialLedger, voteOptions: initialVoteOptions, botDeliveries: initialBotDeliveries,
  };
}

function loadState(): PersistedState {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return freshState();
    const parsed = JSON.parse(raw) as Partial<PersistedState>;
    return { ...freshState(), ...parsed };
  } catch {
    return freshState();
  }
}

const nowLabel = () => new Intl.DateTimeFormat('zh-TW', { month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit', hour12: false }).format(new Date());

export function AppProvider({ children }: { children: ReactNode }) {
  const [state, setState] = useState<PersistedState>(loadState);
  const [toasts, setToasts] = useState<ToastMessage[]>([]);

  useEffect(() => {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state)); } catch { /* 畫面操作仍保留於記憶體 */ }
  }, [state]);

  const showToast = useCallback((text: string, tone: ToastMessage['tone'] = 'success') => {
    const id = Date.now() + Math.random();
    setToasts((items) => [...items, { id, text, tone }]);
    window.setTimeout(() => setToasts((items) => items.filter((item) => item.id !== id)), 3600);
  }, []);

  const currentUser = state.users.find((user) => user.id === state.currentUserId) ?? state.users[0];
  const savedSelections = currentUser.selections;

  const setDraftSelections = (draftSelections: Selection[]) => setState((s) => ({ ...s, draftSelections }));
  const saveSelections = (selections: Selection[]) => setState((s) => ({
    ...s,
    draftSelections: selections,
    users: s.users.map((user) => user.id === s.currentUserId ? { ...user, selections } : user),
  }));

  const updateActivity = (id: string, updater: (activity: Activity) => Activity) => setState((s) => ({
    ...s, activities: s.activities.map((activity) => activity.id === id ? updater(activity) : activity),
  }));

  const createActivity = (activity: Activity) => setState((s) => ({ ...s, activities: [activity, ...s.activities] }));

  const duplicateActivity = (id: string) => {
    const source = state.activities.find((activity) => activity.id === id);
    if (!source) return '';
    const newId = `a${Date.now()}`;
    const copy: Activity = {
      ...source, id: newId, name: `${source.name}（複製）`, status: '草稿', participants: [], rooms: [], version: 1,
      history: [{ time: nowLabel(), text: '複製活動安排；未複製人員、款項或歷史' }],
    };
    setState((s) => ({ ...s, activities: [copy, ...s.activities] }));
    return newId;
  };

  const register: AppContextValue['register'] = (activityId, people, waitlistMode) => {
    setState((s) => ({
      ...s,
      activities: s.activities.map((activity) => {
        if (activity.id !== activityId) return activity;
        const existingActive = activity.participants.filter((p) => ['正式', '待確認'].includes(p.status)).length;
        const created: Participant[] = [];
        let cursor = existingActive;
        people.forEach((person, index) => {
          const isFull = cursor >= activity.capacity;
          const status: Participant['status'] = isFull ? '候補' : '正式';
          cursor += 1;
          created.push({
            id: `p${Date.now()}-${index}`, ...person, status, attendance: '未簽到', paid: 0, createdAt: nowLabel(),
          });
        });
        const hasWaitlist = created.some((p) => p.status === '候補');
        if (hasWaitlist && waitlistMode === '整組候補') created.forEach((p) => { p.status = '候補'; });
        return {
          ...activity,
          status: activity.participants.filter((p) => p.status === '正式').length + created.filter((p) => p.status === '正式').length >= activity.capacity ? '已額滿' : activity.status,
          participants: [...activity.participants, ...created],
          history: [...activity.history, { time: nowLabel(), text: `${currentUser.name} 提交 ${created.length} 人報名（${waitlistMode}）` }],
        };
      }),
      notifications: [{ id: `n${Date.now()}`, title: '報名已提交', body: `已收到 ${people.length} 人報名，費用仍以待結算帳單為準。`, time: '剛剛', read: false, todo: false, done: false, type: '活動', link: `/activities/${activityId}` }, ...s.notifications],
    }));
  };

  const updateParticipant = (activityId: string, participantId: string, patch: Partial<Participant>) => updateActivity(activityId, (activity) => ({
    ...activity,
    participants: activity.participants.map((p) => p.id === participantId ? { ...p, ...patch } : p),
    history: [...activity.history, { time: nowLabel(), text: `更新 ${activity.participants.find((p) => p.id === participantId)?.name ?? '成員'} 的報名／出席資料` }],
  }));

  const updateUser = (id: string, patch: Partial<User>) => setState((s) => ({ ...s, users: s.users.map((user) => user.id === id ? { ...user, ...patch } : user) }));
  const switchUser = (id: string) => setState((s) => ({ ...s, currentUserId: id, draftSelections: s.users.find((user) => user.id === id)?.selections ?? [] }));
  const saveTerm = (term: Term) => setState((s) => ({ ...s, terms: s.terms.some((item) => item.id === term.id) ? s.terms.map((item) => item.id === term.id ? term : item) : [term, ...s.terms] }));
  const deleteTerm = (id: string) => setState((s) => ({ ...s, terms: s.terms.filter((term) => term.id !== id), users: s.users.map((user) => ({ ...user, selections: user.selections.filter((selection) => selection.termId !== id) })), draftSelections: s.draftSelections.filter((selection) => selection.termId !== id) }));
  const updateNotification = (id: string, patch: Partial<AppNotification>) => setState((s) => ({ ...s, notifications: s.notifications.map((n) => n.id === id ? { ...n, ...patch } : n) }));

  const declareBill = (id: string) => setState((s) => ({
    ...s,
    bills: s.bills.map((bill) => bill.id === id ? { ...bill, status: '待核對', declaredAt: '剛剛', method: '模擬轉帳申報' } : bill),
    ledger: [{ id: `l${Date.now()}`, time: nowLabel(), type: '付款申報', person: currentUser.name, amount: s.bills.find((b) => b.id === id)?.due ?? 0, note: '僅為付款申報，尚未核對到帳', verified: false }, ...s.ledger],
  }));

  const verifyBill = (id: string) => setState((s) => ({
    ...s,
    bills: s.bills.map((bill) => bill.id === id ? { ...bill, status: '已結清' } : bill),
    ledger: s.ledger.map((entry, index) => index === 0 && !entry.verified ? { ...entry, verified: true, note: '結算人已核對（模擬）' } : entry),
  }));

  const setVote = (optionId: string, value: '可以' | '可能' | '不行') => setState((s) => ({
    ...s, voteOptions: s.voteOptions.map((option) => option.id === optionId ? { ...option, votes: { ...option.votes, [s.currentUserId]: value } } : option),
  }));
  const finalizeVote = (optionId: string) => setState((s) => ({ ...s, voteOptions: s.voteOptions.map((option) => ({ ...option, final: option.id === optionId })) }));
  const retryDelivery = (id: string) => setState((s) => ({ ...s, botDeliveries: s.botDeliveries.map((delivery) => delivery.id === id ? { ...delivery, status: '已投遞（模擬）', error: undefined, time: '剛剛' } : delivery) }));

  const resetPrototype = () => {
    localStorage.removeItem(STORAGE_KEY);
    setState(freshState());
    showToast('已恢復初始示例資料', 'info');
  };

  const value = useMemo<AppContextValue>(() => ({
    ...state, currentUser, savedSelections, toasts, showToast,
    dismissToast: (id) => setToasts((items) => items.filter((item) => item.id !== id)),
    setDraftSelections, saveSelections, updateActivity, createActivity, duplicateActivity, register,
    updateParticipant, updateUser, switchUser, saveTerm, deleteTerm, updateNotification, declareBill, verifyBill, setVote, finalizeVote, retryDelivery, resetPrototype,
  }), [state, currentUser, savedSelections, toasts, showToast]);

  return <AppContext.Provider value={value}>{children}</AppContext.Provider>;
}

export function useApp() {
  const context = useContext(AppContext);
  if (!context) throw new Error('useApp must be used within AppProvider');
  return context;
}

export const mockService = {
  write: async <T,>(value: T, shouldFail = false): Promise<T> => {
    await new Promise((resolve) => window.setTimeout(resolve, 420));
    if (shouldFail) throw new Error('模擬寫入未完成，原稿仍保留，請重試。');
    return value;
  },
};
